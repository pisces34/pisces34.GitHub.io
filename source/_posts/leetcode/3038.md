### Java
简单模拟
``` Java
class Solution {
    public int maxOperations(int[] nums) {
        int sum = nums[0] + nums[1];
        int count = 1;
        for (int i = 2; i + 1 < nums.length; i++) {
            if (nums[i] + nums[i+1] == sum) {
                count ++;
                i++;
            } else {
                break;
            }
        }
        return count;
    }
}
```
