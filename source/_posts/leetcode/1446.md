``` Java
class Solution {
    public int maxPower(String s) {
        int n = s.length();
        if (n == 1) return 1;
        int res = 0;
        for (int i = 0; i < n - 1; i++) {
            int start = i;
            char x = s.charAt(i);
            while (i < n - 1 && x == s.charAt(i + 1)) {
                i++;
            }
            res = Math.max(res, i-start+1);
        }

        return res;
    }
}
```
