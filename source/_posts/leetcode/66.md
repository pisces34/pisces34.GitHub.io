``` Java
class Solution {
public int[] plusOne(int[] digits) {
        int last = digits.length - 1;
        for (int i = last; i >= 0; i--) {
            if (digits[i] + 1 == 10) { // 有进位
                digits[i] = 0;
            } else {
                digits[i] += 1;
                return digits;
            }
        }
        // 进位到第一位发生溢出，例如[9] 返回 [1,0]
        int[] overflow = new int[digits.length + 1];
        overflow[0] = 1;
        return overflow;
    }
}
```
