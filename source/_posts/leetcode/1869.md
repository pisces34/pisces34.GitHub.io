``` JavaScript
/**
 * @param {string} s
 * @return {boolean}
 */
var checkZeroOnes = function(s) {

    let maxZero = 0
    let maxOne = 0
    const n = s.length
    for (let i = 0; i < n; i++) {
        let zero = 0
        let one = 0
        let j = i
        let k = i
        while (j < n && s[j] === '1') {
            one ++
            j++
        }
        while (k < n && s[k] === '0') {
            zero ++
            k++
        }
        if (one > maxOne) {
            maxOne = one
        }
        if (zero > maxZero) {
            maxZero = zero
        }
    }
    if (maxOne > maxZero) {
        return true
    }
    return false
};
```
